cmake_minimum_required(VERSION 3.24)
project(TerrainDependencyComposition LANGUAGES CXX)
set(TC_TEST_COMPOSITION alone CACHE STRING "alone, terrain-first, or scene-first")
set_property(CACHE TC_TEST_COMPOSITION PROPERTY STRINGS alone terrain-first scene-first)
if(NOT TC_TEST_COMPOSITION MATCHES "^(alone|terrain-first|scene-first)$")
    message(FATAL_ERROR "Unknown dependency composition")
endif()
set(SCENE_POLYTREE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SCENE_POLYTREE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
# Negative fixtures verify that unrelated pre-existing targets cannot bypass pins.
if(TC_TEST_CONFLICT_POLYTREE)
    add_library(conflicting_polytree INTERFACE)
    add_library(polytree::polytree ALIAS conflicting_polytree)
endif()
if(TC_TEST_CONFLICT_ALGO)
    add_library(conflicting_algo INTERFACE)
    add_library(algo::algo ALIAS conflicting_algo)
endif()
get_filename_component(repository "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(helper "${repository}/external/polytree/external/algo/cmake/GitSubmodule.cmake")
if(NOT EXISTS "${helper}")
    message(FATAL_ERROR "Missing dependency submodules. Run git submodule update --init --recursive")
endif()
include("${helper}")
if(TC_TEST_COMPOSITION STREQUAL "scene-first")
    wz_add_git_submodule("${repository}" tests/dependencies/scene-polytree scene-polytree::core)
endif()
add_subdirectory(../coverage coverage)
if(TC_TEST_COMPOSITION STREQUAL "terrain-first")
    wz_add_git_submodule("${repository}" tests/dependencies/scene-polytree scene-polytree::core)
endif()
if(NOT TC_TEST_COMPOSITION STREQUAL "alone")
    target_link_libraries(terrain_coverage_tests PRIVATE scene-polytree::core)
    target_sources(terrain_coverage_tests PRIVATE SceneCompatibility.cpp)
endif()
if(TARGET polytree_tests OR TARGET algo_tests OR TARGET polytree_test_core OR TARGET algo_test_next)
    message(FATAL_ERROR "Dependency tests leaked into the consumer build")
endif()
